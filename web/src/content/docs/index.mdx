import { Card, CardGrid } from '@astrojs/starlight/components';
import AsciinemaPlayer from '../../components/AsciinemaPlayer.astro';
import { CATALOG, TOPICS } from '../../data/catalog';

export const exerciseCount = CATALOG.length;
export const topicCount = Object.keys(TOPICS).length;

export const STAGE_BLURBS = {
  Beginner: 'Syntax you cannot skip: values, control flow, and the built-in collections.',
  Intermediate: 'The type system Go programs are actually shaped from, plus errors and generics.',
  Advanced: 'Goroutines, channels, and the stdlib corners real services live in.',
  Capstone: 'One concurrent log-ingest service, built from everything above it.',
};

export const stageOf = (topic) => TOPICS[topic].tier.split('·')[0].trim();
export const groupOf = (topic) => TOPICS[topic].tier.split('·')[1].trim();

export const stages = Object.keys(STAGE_BLURBS).map((stage) => {
  const topics = Object.keys(TOPICS).filter((t) => stageOf(t) === stage);
  return {
    stage,
    groups: [...new Set(topics.map(groupOf))],
    topicCount: topics.length,
    exercises: CATALOG.filter((e) => stageOf(e.topic) === stage).length,
  };
});

## What is golings?

golings is [rustlings](https://rustlings.rust-lang.org/) for Go — the same idea,
a different language. If you learn a language by debugging it rather than by
reading about it, this is the whole course: open the next exercise, make it
compile, make it pass, move on.

## The path, stage by stage

Four stages, in order. You start at `var x int` and finish with a concurrent
log-ingest service you wrote yourself.
